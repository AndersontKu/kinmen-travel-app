🛰️ GPS 定位邏輯說明
App 利用 Haversine 公式 計算當前座標與所有行程節點之距離，找出最近景點並顯示至下一站資訊。
// GPS 距離計算核心邏輯 function haversine(lat1,lon1,lat2,lon2){ const R = 6371; // 地球半徑 km const dLat = toRad(lat2-lat1); const dLon = toRad(lon2-lon1); const a = Math.sin(dLat/2)**2 + Math.cos(toRad(lat1)) * Math.cos(toRad(lat2)) * Math.sin(dLon/2)**2; return R * 2 * Math.atan2( Math.sqrt(a), Math.sqrt(1-a) ); } // 找最近景點 function findNearest(userLat,userLon){ return spots.reduce((nearest,spot)=>{ const d = haversine( userLat,userLon, spot.lat,spot.lon ); return d < nearest.dist ? {spot,dist:d} : nearest; },{dist:Infinity}); } // 機車時速換算 (均速 30km/h) const motoTime = (km) => Math.round(km/30*60); // 分鐘
