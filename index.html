// GPS 距離計算核心邏輯 function haversine(lat1,lon1,lat2,lon2){ const R = 6371; // 地球半徑 km const dLat = toRad(lat2-lat1); const dLon = toRad(lon2-lon1); const a = Math.sin(dLat/2)**2 + Math.cos(toRad(lat1)) * Math.cos(toRad(lat2)) * Math.sin(dLon/2)**2; return R * 2 * Math.atan2( Math.sqrt(a), Math.sqrt(1-a) ); } // 機車時速換算 (均速 30km/h) const motoTime = (km) => Math.round(km/30*60); // 分鐘
